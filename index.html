// فەنکشنی دروستکردنی PDFی ڕاستکراوە function generatePDF(data) { // دروستکردنی کۆنتێنەرێکی کاتی بۆ PDF const element = document.createElement('div'); element.style.position = 'absolute'; element.style.left = '-9999px'; element.style.width = '800px'; element.style.padding = '30px'; element.style.backgroundColor = '#ffffff'; element.style.color = '#000000'; element.style.direction = 'rtl'; element.style.fontFamily = 'Tahoma, Arial, sans-serif'; // ناوەرۆکی ڕاپۆرتەکە element.innerHTML = `

قوتابخانەی گەیلانی بۆ زانستە شرعییەکان

فۆڕمی تۆمارکردنی قوتابی (٢٠٢٦-٢٠٢٧)

ناوی تەواو: ${data.fullName || ''}
بەرواری لەدایکبوون: ${data.birthDate || ''}
پڕۆفێشن / پێشە: ${data.profession || ''}
ژمارەی تلفۆن: ${data.phone || ''}
ئاستی خوێندن: ${data.education || ''}

بەرواری تۆمارکردن: ${new Date().toLocaleDateString('en-GB')}

`; document.body.appendChild(element); const opt = { margin: 10, filename: `Gaylani_Student_${data.fullName || 'Report'}.pdf`, image: { type: 'jpeg', quality: 0.98 }, html2canvas: { scale: 2, useCORS: true, logging: false }, jsPDF: { unit: 'mm', format: 'a4', orientation: 'portrait' } }; // ڕەندەرکردن و دابەزاندن html2pdf().set(opt).from(element).save().then(() => { document.body.removeChild(element); }); }
